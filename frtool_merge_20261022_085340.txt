num toNum(dynamic v) {
  if (v is num) return v;
  if (v is String) return num.tryParse(v) ?? 0;
  return 0;
}

class Aktivitas {
  Aktivitas({
    required this.id,
    required this.col,
    required this.tipe,
    required this.arus,
    required this.keterangan,
    required this.tanggal,
    required this.tempat,
    required this.tujuan,
    required this.nominal,
    required this.harga,
    required this.waktu,
    required this.siklusAktif,
  });

  final String id, col, tipe, arus, keterangan, tanggal, tempat, tujuan;
  final num nominal, harga;
  final DateTime waktu;
  final bool siklusAktif;

  bool get isMasuk => arus == 'masuk';
  bool get isKeluar => arus == 'keluar';
  bool get isUang => col == 'aktivitas_dompeng';

  factory Aktivitas.fromDoc(String col, QueryDocumentSnapshot<Map<String, dynamic>> d) {
    final m = d.data();
    final ts = m['createdAt'];
    return Aktivitas(
      id: d.id,
      col: col,
      tipe: (m['tipe'] ?? '').toString(),
      arus: (m['arus'] ?? '').toString(),
      keterangan: (m['keterangan'] ?? '').toString(),
      tanggal: (m['tanggal'] ?? '').toString(),
      tempat: (m['tempat'] ?? '').toString(),
      tujuan: (m['tujuan'] ?? '').toString(),
      nominal: toNum(m['nominal']),
      harga: toNum(m['harga']),
      waktu: ts is Timestamp ? ts.toDate() : DateTime.fromMillisecondsSinceEpoch(0),
      siklusAktif: m['isSiklusAktif'] != false,
    );
  }
}

/// Sama persis dengan hitungBiayaBakar() di stok-tab.js
Map<String, dynamic> hitungBiayaBakar(num jumlah, [num karung = 0, num hargaMerang = 0]) {
  final dompeng = (jumlah / 1000) * 60000;
  final cetak = (jumlah / 1000) * 80000;
  final tanahMobil = (jumlah / 5000).ceil();
  final tanah = tanahMobil * 270000;
  final merang = karung * hargaMerang;
  return {
    'dompeng': dompeng,
    'cetak': cetak,
    'tanahMobil': tanahMobil,
    'tanah': tanah,
    'karungMerang': karung,
    'hargaMerang': hargaMerang,
    'merang': merang,
    'total': dompeng + cetak + tanah + merang,
  };
}

  void start() {

  void _state(String id, void Function(Map<String, dynamic>) apply) {
    _subs.add(_db.collection('app_state').doc(id).snapshots().listen((s) {
      apply(s.data() ?? <String, dynamic>{});
      _changed();
    }, onError: _onErr));
  }

  void _col(String col, int? limit, void Function(List<Aktivitas>) apply) {
    Query<Map<String, dynamic>> q = _db.collection(col).orderBy('createdAt', descending: true);
    if (limit != null) q = q.limit(limit);
    _subs.add(q.snapshots().listen((s) {
      apply(s.docs.map((d) => Aktivitas.fromDoc(col, d)).toList());
      _changed();
    }, onError: _onErr));
  }

    _semua = null;

  /// Siklus = semua transaksi dompeng dengan isSiklusAktif != false
  num get siklus {
    num t = 0;
    for (final a in dompeng) {
      if (!a.siklusAktif) continue;
      if (a.isMasuk) {
        t += a.nominal;
      } else if (a.isKeluar) {
        t -= a.nominal;
      }
    }
    return t;
  }

  bool _uang(Aktivitas a) => a.tipe != 'Kirim ke Total Utama' && a.tipe != 'Penyesuaian Saldo';

  String get _bulan => tanggalHariIni().substring(0, 7);

  num get pemasukanBulan => dompeng
      .where((a) => a.isMasuk && _uang(a) && a.tanggal.startsWith(_bulan))
      .fold<num>(0, (t, a) => t + a.nominal);

  num get pengeluaranBulan => dompeng
      .where((a) => a.isKeluar && _uang(a) && a.tanggal.startsWith(_bulan))
      .fold<num>(0, (t, a) => t + a.nominal);

    if (f == 1) return dompeng.take(150).toList();
    _semua ??= _gabung();
    if (f == 2) return _semua!.where((a) => !a.isUang).toList();
    return _semua!;
  }

  List<Aktivitas> _gabung() {
    // sisi kembar dari satu transaksi disembunyikan supaya tidak dobel
    const sembunyi = {'Bakar Bata', 'Batu Siap Jual', 'Penjualan Bata'};
    final l = <Aktivitas>[...dompeng.take(120), ...mentah, ...beranda, ...siapjual]
        .where((a) => !sembunyi.contains(a.tipe))
        .toList();
    l.sort((a, b) => b.waktu.compareTo(a.waktu));
    return l.take(200).toList();
  }

  DocumentReference<Map<String, dynamic>> _st(String id) => _db.collection('app_state').doc(id);
  DocumentReference<Map<String, dynamic>> _ref(String col, String id) => _db.collection(col).doc(id);
  String _id(String p) => '${p}_${DateTime.now().microsecondsSinceEpoch}';

  /// Tidak di-await supaya jalan offline-first seperti versi web
  void _go(WriteBatch b) {
    unawaited(b.commit().catchError((Object e) {
      error = e.toString();

  }

  Map<String, dynamic> _akt({
    required String tipe,
    required String arus,
    required num nominal,
    String ket = '',
    Map<String, dynamic> extra = const {},

    return {
      'tipe': tipe,
      'arus': arus,
      'nominal': nominal,
      'keterangan': ket,
      'tanggal': tanggalHariIni(),
      'createdAt': Timestamp.now(),
      ...extra,
    };
  }

  // ───────────── WORKFLOW (return null = sukses, selain itu pesan error) ─────────────

  /// 1. Cetak bata -> stok mentah (kulon / etan)
  String? cetakBata({required String tempat, required int jumlah, String ket = ''}) {
    if (jumlah <= 0) return 'Jumlah bata harus lebih dari 0';
    final b = _db.batch();
    b.set(_st('mentah'), {tempat: FieldValue.increment(jumlah)}, SetOptions(merge: true));
    b.set(
      _ref('aktivitas_mentah', _id('cb')),
      _akt(tipe: 'Cetak Bata', arus: 'masuk', nominal: jumlah, ket: ket, extra: {'tempat': tempat}),
    );
    _go(b);
    return null;
  }

  /// 2. Mentah -> tungku (Total Stok Batu)
  String? pindahKeTungku({required String tempat, required int jumlah, String ket = ''}) {
    final stok = tempat == 'kulon' ? mentahKulon : mentahEtan;
    if (jumlah <= 0) return 'Jumlah bata harus lebih dari 0';
    if (jumlah > stok) return 'Stok mentah ${areaLabel(tempat)} hanya ${fmt(stok)} bata';
    final id = _id('bkr');
    final b = _db.batch();
    b.set(_st('mentah'), {tempat: FieldValue.increment(-jumlah)}, SetOptions(merge: true));
    b.set(_st('batu'), {'total': FieldValue.increment(jumlah)}, SetOptions(merge: true));
    b.set(
      _ref('aktivitas_mentah', id),
      _akt(tipe: 'Bakar Bata', arus: 'keluar', nominal: jumlah, ket: ket, extra: {'tempat': tempat}),
    );
    b.set(
      _ref('aktivitas_beranda', id),
      _akt(
        tipe: 'Bata Mentah Masuk',
        arus: 'masuk',
        nominal: jumlah,
        ket: ket.isEmpty ? 'Dari mentah ${areaLabel(tempat)}' : ket,
      ),
    );
    _go(b);
    return null;
  }

  /// 3a. Mulai sesi bakar
  String? mulaiBakar() {
    if (burning) return 'Masih ada sesi bakar yang berjalan';
    if (batu <= 0) return 'Stok di tungku masih kosong';
    final n = sesiCount + 1;
    final id = _id('bkr');
    final t = tanggalHariIni();
    final Map<String, dynamic> item = {
      'id': id,
      'sesi': n,
      'nominal': batu,
      'tanggal': t,
      'tanggalMulai': t,
      'biaya': hitungBiayaBakar(batu),
    };
    final b = _db.batch();
    b.set(
      _st('bakar'),
      {
        'sesiCount': n,
        'status': 'burning',
        'riwayatSesi': [...sesi, item],
      },
      SetOptions(merge: true),
    );
    b.set(
      _ref('aktivitas_beranda', id),
      _akt(tipe: 'Mulai Bakar', arus: 'netral', nominal: batu, ket: 'Sesi $n'),
    );
    _go(b);
    return null;
  }

  /// 3b. Selesai bakar (biaya merang opsional)
  String? selesaiBakar({int karung = 0, int hargaMerang = 0}) {
    if (sesiAktif == null) return 'Tidak ada sesi bakar yang berjalan';
    final list = sesi.map((e) => Map<String, dynamic>.from(e)).toList();
    final last = list.last;
    final nominal = toNum(last['nominal']);
    final sn = last['sesi'];
    last['selesai'] = true;
    last['tanggalSelesai'] = tanggalHariIni();
    last['biaya'] = hitungBiayaBakar(nominal, karung, hargaMerang);
    final b = _db.batch();
    b.set(_st('bakar'), {'status': 'idle', 'riwayatSesi': list}, SetOptions(merge: true));
    b.set(
      _ref('aktivitas_beranda', _id('bkr')),
      _akt(tipe: 'Batu Selesai Dibakar', arus: 'masuk', nominal: nominal, ket: 'Sesi $sn'),
    );
    _go(b);
    return null;
  }

  /// 4. Tungku -> gudang siap jual
  String? susunSiapJual({required int jumlah, String ket = ''}) {
    if (jumlah <= 0) return 'Jumlah bata harus lebih dari 0';
    if (jumlah > batu) return 'Stok di tungku hanya ${fmt(batu)} bata';
    final id = _id('bkr');
    final b = _db.batch();
    b.set(_st('batu'), {'total': FieldValue.increment(-jumlah)}, SetOptions(merge: true));
    b.set(_st('siapjual'), {'total': FieldValue.increment(jumlah)}, SetOptions(merge: true));
    b.set(
      _ref('aktivitas_beranda', id),
      _akt(
        tipe: 'Batu Telah Disusun',
        arus: 'keluar',
        nominal: jumlah,
        ket: ket.isEmpty ? 'Disusun ke gudang Siap Jual' : ket,
      ),
    );
    b.set(
      _ref('aktivitas_siapjual', id),
      _akt(
        tipe: 'Batu Siap Jual',
        arus: 'masuk',
        nominal: jumlah,
        ket: ket.isEmpty ? 'Masuk dari pembakaran' : ket,
      ),
    );
    _go(b);
    return null;
  }

  /// 5. Kirim / jual bata -> pendapatan masuk siklus dompeng
  String? kirimBata({required String tujuan, required int jumlah, required int harga, String ket = ''}) {
    if (tujuan.trim().isEmpty) return 'Isi tujuan pengiriman';
    if (jumlah <= 0) return 'Jumlah bata harus lebih dari 0';
    if (jumlah > siapJual) return 'Stok siap jual hanya ${fmt(siapJual)} bata';
    final detail = ket.isEmpty ? '$tujuan • ${fmt(jumlah)} bata' : ket;
    final b = _db.batch();
    b.set(_st('siapjual'), {'total': FieldValue.increment(-jumlah)}, SetOptions(merge: true));
    b.set(
      _ref('aktivitas_siapjual', _id('krm')),
      _akt(
        tipe: 'Pengiriman Bata',
        arus: 'keluar',
        nominal: jumlah,
        ket: detail,
        extra: {'harga': harga, 'tujuan': tujuan.trim()},
      ),
    );
    if (harga > 0) {
      b.set(
        _ref('aktivitas_dompeng', _id('jual')),
        _akt(
          tipe: 'Penjualan Bata',
          arus: 'masuk',
          nominal: jumlah * harga,
          ket: detail,
          extra: {'isSiklusAktif': true},
        ),
      );
    }
    _go(b);
    return null;
  }

  /// Catat pengeluaran (tidak masuk hitungan siklus)
  String? catatPengeluaran({required String kategori, required int nominal, String ket = ''}) {
    if (nominal <= 0) return 'Nominal harus lebih dari 0';
    final b = _db.batch();
    b.set(
      _ref('aktivitas_dompeng', _id('out')),
      _akt(tipe: kategori, arus: 'keluar', nominal: nominal, ket: ket, extra: {'isSiklusAktif': false}),
    );
    _go(b);
    return null;
  }

  /// Kirim ke Total Utama: siklus ditutup dan dijumlahkan ke total_utama
  String? kirimKeUtama() {
    final aktif = dompeng.where((a) => a.siklusAktif).toList();
    if (aktif.isEmpty) return 'Belum ada transaksi di siklus ini';
    final sk = siklus;
    final num baru = totalUtama + sk < 0 ? 0 : totalUtama + sk;
    final jumlahMasuk = aktif.where((a) => a.isMasuk).length;
    var b = _db.batch();
    var ops = 0;
    for (final a in aktif) {
      b.update(_ref('aktivitas_dompeng', a.id), {'isSiklusAktif': false});
      ops++;
      if (ops >= 400) {
        _go(b);
        b = _db.batch();
        ops = 0;
      }
    }
    b.set(_st('dompeng'), {'total_utama': baru}, SetOptions(merge: true));
    b.set(
      _ref('aktivitas_dompeng', _id('krm')),
      _akt(
        tipe: 'Kirim ke Total Utama',
        arus: sk >= 0 ? 'masuk' : 'keluar',
        nominal: sk.abs(),
        ket: '$jumlahMasuk dompengan dikirim',
        extra: {'isPermanent': true, 'isSiklusAktif': false},
      ),
    );
    _go(b);
    return null;
  }
}

class Counter extends StatefulWidget {
  const Counter({
    super.key,
    required this.value,
    required this.style,
    this.symbol = '',
    this.hidden = false,
  });

  final num value;
  final TextStyle style;
  final String symbol;
  final bool hidden;

  @override
  State<Counter> createState() => _CounterState();
}

class _CounterState extends State<Counter> {
  num _shown = 0;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (mounted) setState(() => _shown = widget.value);
    });
  }

  @override
  void didUpdateWidget(covariant Counter oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.value != widget.value) _shown = widget.value;
  }

  @override
  Widget build(BuildContext context) {
    final s = StoreScope.of(context);
    final Widget number = s.counterAnim
        ? RepaintBoundary(
            child: NumberFlow(
              value: _shown,
              locale: 'id_ID',
              format: NumberFlowFormat.currency(currencyCode: 'IDR', symbol: widget.symbol),
              spring: NumberFlowSpring.bouncy,
              motionBlur: 5.5,
              digitSquish: 0.32,
              stagger: const Duration(milliseconds: 40),
              transformTiming: const TimingConfig(duration: Duration(milliseconds: 450), curve: Curves.easeInOut),
              opacityTiming: const TimingConfig(duration: Duration(milliseconds: 450), curve: Curves.easeOut),
              tabularNums: true,
              style: widget.style,
            ),
          )
        : Text('${widget.symbol}${fmt(widget.value)}', style: widget.style);

    return TweenAnimationBuilder<double>(
      tween: Tween<double>(end: widget.hidden ? 12.0 : 0.0),
      duration: const Duration(milliseconds: 320),
      curve: Curves.easeOutCubic,
      builder: (context, sigma, child) => sigma < 0.05
          ? child!
          : ImageFiltered(
              imageFilter: ImageFilter.blur(sigmaX: sigma, sigmaY: sigma, tileMode: TileMode.decal),
              child: child,
            ),
      child: number,
    );
  }
}

/// Buka halaman fullscreen dengan transisi geser + fade
void openPage(BuildContext context, Widget page) {
  Navigator.of(context).push(
    PageRouteBuilder<void>(
      transitionDuration: const Duration(milliseconds: 320),
      reverseTransitionDuration: const Duration(milliseconds: 260),
      pageBuilder: (_, __, ___) => page,
      transitionsBuilder: (_, a, __, child) {
        final c = CurvedAnimation(parent: a, curve: Curves.easeOutCubic, reverseCurve: Curves.easeInCubic);
        return FadeTransition(
          opacity: c,
          child: SlideTransition(
            position: Tween<Offset>(begin: const Offset(0.08, 0), end: Offset.zero).animate(c),
            child: child,
          ),
        );
      },
    ),
  );
}

/// Sheet / dialog dengan latar yang ter-blur (seperti pada desain)
Future<T?> _blurRoute<T>(BuildContext context, {required Widget child, required bool sheet}) {
  return showGeneralDialog<T>(
    context: context,
    barrierDismissible: true,
    barrierLabel: 'Tutup',
    barrierColor: Colors.black.withValues(alpha: 0.12),
    transitionDuration: const Duration(milliseconds: 340),
    pageBuilder: (c, a1, a2) {
      final page = Material(type: MaterialType.transparency, child: child);
      if (sheet) {
        return Align(
          alignment: Alignment.bottomCenter,
          child: Padding(padding: EdgeInsets.only(bottom: MediaQuery.viewInsetsOf(c).bottom), child: page),
        );
      }
      return Center(child: page);
    },
    transitionBuilder: (c, anim, a2, page) {
      final curve = CurvedAnimation(parent: anim, curve: Curves.easeOutCubic, reverseCurve: Curves.easeInCubic);
      return Stack(
        children: [
          Positioned.fill(
            child: AnimatedBuilder(
              animation: curve,
              builder: (context, _) {
                final v = 14 * curve.value;
                return BackdropFilter(
                  filter: ImageFilter.blur(sigmaX: v, sigmaY: v),
                  child: const SizedBox.expand(),
                );
              },
            ),
          ),
          if (sheet)
            SlideTransition(
              position: Tween<Offset>(begin: const Offset(0, 1), end: Offset.zero).animate(curve),
              child: page,
            )
          else
            FadeTransition(
              opacity: curve,
              child: ScaleTransition(scale: Tween<double>(begin: 0.92, end: 1).animate(curve), child: page),
            ),
        ],
      );
    },
  );
}

Future<void> openTx(BuildContext context, TxKind kind, {String? tempat}) async {
  final msg = await _blurRoute<String>(context, sheet: true, child: TxSheet(kind: kind, tempat: tempat));
  if (msg != null && context.mounted) {
    await _blurRoute<void>(context, sheet: false, child: _SuccessCard(message: msg));
  }
}

class ThousandsFormatter extends TextInputFormatter {
  @override
  TextEditingValue formatEditUpdate(TextEditingValue oldValue, TextEditingValue newValue) {
    final digits = newValue.text.replaceAll(RegExp(r'[^0-9]'), '');
    if (digits.isEmpty) return const TextEditingValue(text: '');
    final t = fmt(int.parse(digits));
    return TextEditingValue(text: t, selection: TextSelection.collapsed(offset: t.length));
  }
}

class TxSheet extends StatefulWidget {
  const TxSheet({super.key, required this.kind, this.tempat});

  final TxKind kind;
  final String? tempat;

  @override
  State<TxSheet> createState() => _TxSheetState();
}

class _TxSheetState extends State<TxSheet> {
  final _jumlah = TextEditingController();
  final _harga = TextEditingController();
  final _tujuan = TextEditingController();
  final _ket = TextEditingController();
  final _nominal = TextEditingController();
  final _karung = TextEditingController();
  final _hargaMerang = TextEditingController();

  late String _tempat = widget.tempat ?? 'kulon';
  String _kat = 'Solar';
  late int _step = _hasFields ? 0 : 1;
  String? _err;

  bool get _hasFields => widget.kind != TxKind.mulaiBakar && widget.kind != TxKind.kirimUtama;

  @override
  void dispose() {
    _jumlah.dispose();
    _harga.dispose();
    _tujuan.dispose();
    _ket.dispose();
    _nominal.dispose();
    _karung.dispose();
    _hargaMerang.dispose();
    super.dispose();
  }

  int _int(TextEditingController c) => int.tryParse(c.text.replaceAll('.', '')) ?? 0;

  String get _title => switch (widget.kind) {
        TxKind.cetak => 'Cetak Bata',
        TxKind.pindah => 'Pindah ke Tungku',
        TxKind.mulaiBakar => 'Mulai Bakar',
        TxKind.selesaiBakar => 'Selesai Bakar',
        TxKind.susun => 'Susun ke Siap Jual',
        TxKind.kirim => 'Kirim / Jual Bata',
        TxKind.pengeluaran => 'Catat Pengeluaran',
        TxKind.kirimUtama => 'Kirim ke Total Utama',
      };

  IconData get _icon => switch (widget.kind) {
        TxKind.cetak => Ic.add,
        TxKind.pindah => Ic.fire,
        TxKind.mulaiBakar => Ic.play,
        TxKind.selesaiBakar => Ic.checkLine,
        TxKind.susun => Ic.stock,
        TxKind.kirim => Ic.truck,
        TxKind.pengeluaran => Ic.wallet,
        TxKind.kirimUtama => Ic.send,
      };

  String get _headline => switch (widget.kind) {
        TxKind.cetak => 'Cetak Bata – Stok Mentah',
        TxKind.pindah => 'Bata Mentah – Tungku',
        TxKind.mulaiBakar => 'Sesi Bakar Baru',
        TxKind.selesaiBakar => 'Pembakaran Selesai',
        TxKind.susun => 'Tungku – Gudang Siap Jual',
        TxKind.kirim => 'Pengiriman – Penjualan Bata',
        TxKind.pengeluaran => 'Pengeluaran Usaha',
        TxKind.kirimUtama => 'Tutup Siklus Kas',
      };

  String? _validate(AppStore s) {
    final j = _int(_jumlah);
    switch (widget.kind) {
      case TxKind.cetak:
        return j <= 0 ? 'Isi jumlah bata yang dicetak' : null;
      case TxKind.pindah:
        final stok = _tempat == 'kulon' ? s.mentahKulon : s.mentahEtan;
        if (j <= 0) return 'Isi jumlah bata';
        if (j > stok) return 'Stok mentah ${areaLabel(_tempat)} hanya ${fmt(stok)} bata';
        return null;
      case TxKind.susun:
        if (j <= 0) return 'Isi jumlah bata';
        if (j > s.batu) return 'Stok di tungku hanya ${fmt(s.batu)} bata';
        return null;
      case TxKind.kirim:
        if (_tujuan.text.trim().isEmpty) return 'Isi tujuan pengiriman';
        if (j <= 0) return 'Isi jumlah bata';
        if (j > s.siapJual) return 'Stok siap jual hanya ${fmt(s.siapJual)} bata';
        return null;
      case TxKind.pengeluaran:
        return _int(_nominal) <= 0 ? 'Isi nominal pengeluaran' : null;
      case TxKind.selesaiBakar:
      case TxKind.mulaiBakar:
      case TxKind.kirimUtama:
        return null;
    }
  }

  List<List<String>> _rows(AppStore s) {
    final j = _int(_jumlah);
    final harga = _int(_harga);
    switch (widget.kind) {
      case TxKind.cetak:
        return [
          ['Jenis', 'Cetak bata mentah'],
          ['Area', areaLabel(_tempat)],
          ['Jumlah', '${fmt(j)} bata'],
        ];
      case TxKind.pindah:
        return [
          ['Dari', 'Mentah ${areaLabel(_tempat)}'],
          ['Ke', 'Tungku bakar'],
          ['Jumlah', '${fmt(j)} bata'],
        ];
      case TxKind.mulaiBakar:
        return [
          ['Sesi', 'Sesi ke-${s.sesiCount + 1}'],
          ['Batu di tungku', '${fmt(s.batu)} bata'],
          ['Estimasi biaya', rp(toNum(hitungBiayaBakar(s.batu)['total']))],
        ];
      case TxKind.selesaiBakar:
        final a = s.sesiAktif;
        final n = a == null ? 0 : toNum(a['nominal']);
        return [
          ['Sesi', '${a == null ? '-' : a['sesi']}'],
          ['Batu dibakar', '${fmt(n)} bata'],
          ['Merang', '${_int(_karung)} karung × ${rp(_int(_hargaMerang))}'],
        ];
      case TxKind.susun:
        return [
          ['Dari', 'Tungku bakar'],
          ['Ke', 'Gudang siap jual'],
          ['Jumlah', '${fmt(j)} bata'],
        ];
      case TxKind.kirim:
        return [
          ['Tujuan', _tujuan.text.trim()],
          ['Dari', 'Stok siap jual'],
          ['Jumlah', '${fmt(j)} bata'],
          ['Harga / bata', harga > 0 ? rp(harga) : 'Tanpa harga'],
        ];
      case TxKind.pengeluaran:
        return [
          ['Kategori', _kat],
          ['Catatan', _ket.text.trim().isEmpty ? '-' : _ket.text.trim()],
        ];
      case TxKind.kirimUtama:
        return [
          ['Total utama', rp(s.totalUtama)],
          ['Siklus berjalan', rp(s.siklus)],
        ];
    }
  }

  List<String> _total(AppStore s) {
    final j = _int(_jumlah);
    switch (widget.kind) {
      case TxKind.cetak:
      case TxKind.pindah:
      case TxKind.susun:
        return ['Total', '${fmt(j)} bata'];
      case TxKind.mulaiBakar:
        return ['Batu dibakar', '${fmt(s.batu)} bata'];
      case TxKind.selesaiBakar:
        final a = s.sesiAktif;
        final n = a == null ? 0 : toNum(a['nominal']);
        final t = toNum(hitungBiayaBakar(n, _int(_karung), _int(_hargaMerang))['total']);
        return ['Total biaya', rp(t)];
      case TxKind.kirim:
        final h = _int(_harga);
        return h > 0 ? ['Total', rp(j * h)] : ['Total', '${fmt(j)} bata'];
      case TxKind.pengeluaran:
        return ['Total', rp(_int(_nominal))];
      case TxKind.kirimUtama:
        final baru = s.totalUtama + s.siklus;
        return ['Utama baru', rp(baru < 0 ? 0 : baru)];
    }
  }

  void _next(AppStore s) {
    final e = _validate(s);
    setState(() {
      _err = e;
      if (e == null) _step = 1;
    });
  }

  void _submit(AppStore s) {
    final j = _int(_jumlah);
    final ket = _ket.text.trim();
    String? err;
    String msg;
    switch (widget.kind) {
      case TxKind.cetak:
        err = s.cetakBata(tempat: _tempat, jumlah: j, ket: ket);
        msg = '${fmt(j)} bata mentah area ${areaLabel(_tempat)} sudah dicatat.';
      case TxKind.pindah:
        err = s.pindahKeTungku(tempat: _tempat, jumlah: j, ket: ket);
        msg = '${fmt(j)} bata dipindahkan dari mentah ${areaLabel(_tempat)} ke tungku.';
      case TxKind.mulaiBakar:
        err = s.mulaiBakar();
        msg = 'Sesi bakar dimulai dengan ${fmt(s.batu)} bata di tungku.';
      case TxKind.selesaiBakar:
        err = s.selesaiBakar(karung: _int(_karung), hargaMerang: _int(_hargaMerang));
        msg = 'Sesi bakar ditandai selesai. Bata siap disusun ke gudang.';
      case TxKind.susun:
        err = s.susunSiapJual(jumlah: j, ket: ket);
        msg = '${fmt(j)} bata sudah disusun ke stok siap jual.';
      case TxKind.kirim:
        err = s.kirimBata(tujuan: _tujuan.text, jumlah: j, harga: _int(_harga), ket: ket);
        msg = 'Pengiriman ${fmt(j)} bata ke ${_tujuan.text.trim()} sudah dicatat.';
      case TxKind.pengeluaran:
        err = s.catatPengeluaran(kategori: _kat, nominal: _int(_nominal), ket: ket);
        msg = 'Pengeluaran $_kat sebesar ${rp(_int(_nominal))} sudah dicatat.';
      case TxKind.kirimUtama:
        err = s.kirimKeUtama();
        msg = 'Siklus kas sudah dikirim ke total utama.';
    }
    if (err != null) {
      setState(() {
        _err = err;
        if (_hasFields) _step = 0;
      });
      return;
    }
    Navigator.of(context).pop(msg);
  }

/// 12000 -> "12.000"
String fmt(num n) {
  final neg = n < 0;
  final s = n.abs().round().toString();
  final b = StringBuffer();
  for (var i = 0; i < s.length; i++) {
    if (i > 0 && (s.length - i) % 3 == 0) b.write('.');
    b.write(s[i]);
  }
  return neg ? '-$b' : b.toString();
}

String rp(num n) => n < 0 ? '-Rp ${fmt(n.abs())}' : 'Rp ${fmt(n)}';

String tanggalHariIni([DateTime? d]) {
  final n = d ?? DateTime.now();
  final m = n.month.toString().padLeft(2, '0');
  final h = n.day.toString().padLeft(2, '0');
  return '${n.year}-$m-$h';
}

const _bln = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des'];

String tglCantik(String ymd) {
  final p = ymd.split('-');
  if (p.length != 3) return ymd;
  final m = int.tryParse(p[1]) ?? 1;
  final d = int.tryParse(p[2]) ?? 0;
  return '$d ${_bln[(m - 1).clamp(0, 11).toInt()]}';
}

        child: widget.child,
      ),
    );
